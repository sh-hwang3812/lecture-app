강의 프리랜서 강의 일정 관리

구글 스프레드시트 연동 (Google Apps Script)

상용 유료 DB 없이 구글 시트를 무료 영구 데이터베이스로 활용합니다.

제약조건 보증

• 무료 계정 최적화: 구글 일일 실행 한도를 준수하여 단건 추가 및 단순 조회로 가볍게 작동합니다.
• 개인정보 비저장: 기관명은 가상 별칭만 사용되며 실명/연락처는 시트에 절대 저장되지 않습니다.

연동 방법 (3단계, 1분 소요)
  1. 새 구글 스프레드시트를 생성하고 확장 프로그램 > Apps Script를 엽니다.
  2. 아래 Apps Script 코드를 붙여넣은 뒤 setupSheet 함수를 1회 실행합니다.
  3. 우측 상단 배포 > 새 배포 > 유형: 웹 앱을 선택하고 액세스 권한을 '모든 사용자(Anyone)'로 설정 후 배포된 URL을 위 입력창에 등록합니다.
/**
 * [프리랜서 강의 일정 관리] Google Apps Script
 * 제약조건: 무료 계정 쿼터 최적화, 실명/연락처 배제, 무상태 JSON 통신
 */

function setupSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName('강의일정') || ss.insertSheet('강의일정');
  
  // 8개 시트 열 헤더 (엄격 유지)
  const headers = ['날짜', '기관', '시작', '종료', '시간', '상태', '장소', '메모'];
  sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#f1f5f9');
  
  // 상태 드롭다운 (확정 / 변경 / 대기)
  const rule = SpreadsheetApp.newDataValidation()
    .requireValueInList(['확정', '변경', '대기'], true)
    .build();
  sheet.getRange('F2:F1000').setDataValidation(rule);
  sheet.setFrozenRows(1);
}

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('강의일정') || ss.getSheets()[0];
  const data = sheet.getDataRange().getValues();
  const rows = data.slice(1);
  
  const schedules = rows.map((r, i) => ({
    id: 'sheet-' + (i + 1),
    date: Utilities.formatDate(new Date(r[0]), Session.getScriptTimeZone(), 'yyyy-MM-dd'),
    institution: r[1],
    startTime: r[2],
    endTime: r[3],
    hours: Number(r[4]),
    status: r[5],
    location: r[6],
    notes: r[7] || ''
  }));
  
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    data: schedules
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const item = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName('강의일정') || ss.getSheets()[0];
    
    const [startH, startM] = item.startTime.split(':').map(Number);
    const [endH, endM] = item.endTime.split(':').map(Number);
    const hours = Number((((endH * 60 + endM) - (startH * 60 + startM)) / 60).toFixed(1));
    
    sheet.appendRow([
      item.date,
      item.institution,
      item.startTime,
      item.endTime,
      hours,
      item.status,
      item.location,
      item.notes || ''
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ success: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
